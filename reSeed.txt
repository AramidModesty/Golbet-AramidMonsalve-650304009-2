## To reseed your database, you have to first drop the existing database
##  and then migrate the database again.
## Example of the drop of database:
# // where context is already defined as AppDbContext context
# context.Database.EnsureDeleted();
# context.Database.EnsureCreated();
# DbSeeder.Seed(context);

## If you want to clean a certain table of entities like Matches,
##  you only drop that and then migrate your database
## Example of the drop of entity:
# // where context is already defined as AppDbContext context
# context.Matches.RemoveRange(context.Matches);
# context.SaveChanges();
# DbSeeder.Seed(context);
